local v274

local function v275(p50)
	local HumanoidRootPart = p50:FindFirstChild("HumanoidRootPart")
	local Humanoid = p50:FindFirstChildOfClass("Humanoid")

	if not HumanoidRootPart or not Humanoid then
		return nil
	end

	if Humanoid.FloorMaterial == Enum.Material.Air then
		return nil
	end

	local raycastParams = RaycastParams.new()

	raycastParams.FilterType = Enum.RaycastFilterType.Exclude
	raycastParams.FilterDescendantsInstances = { p50 }

	local hit = workspace:Raycast(HumanoidRootPart.Position, Vector3.new(0, -10, 0), raycastParams)

	if hit then
		if hit.Instance:IsA("BasePart") then
			return HumanoidRootPart.CFrame
		end
	end

	return nil
end

while task.wait(0.1) do
	local Character = LocalPlayer2.Character
	local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

	if HumanoidRootPart then
		local Y = HumanoidRootPart.Position.Y

		if Y < workspace.FallenPartsDestroyHeight or Y > 5000 then
			now = now or os.clock()

			if os.clock() - now >= 5 then
				if v274 then
					HumanoidRootPart.CFrame = v274 + Vector3.new(0, 4, 0)
					HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
					HumanoidRootPart.AssemblyAngularVelocity = Vector3.zero
				end

				now = nil
			end
		else
			v274 = v275(Character) or v274
			now = nil
		end
	else
		now = nil
	end
end
